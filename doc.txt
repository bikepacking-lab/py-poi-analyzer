# Environnement python
# Windows création de l'environnement
python -m venv .venv

# Windows activation de l'environnement
.venv\Scripts\activate

# Installation des dépendances
pip install -r requirements.txt

# Utilisation
# Simple fichier gpx à analyser
python main.py chemin/vers/fichier.gpx

# Avec options
python main.py chemin/vers/fichier.gpx --poi-types water refueling bivouac --report-formats json csv pdf

# Avec paramètres avancés
python main.py track.gpx \
    --radii '{"water":1000,"refueling":1500}' \
    --intervals '{"water":30,"refueling":40}' \
    --debug
